    // Gauge animator

    function updateGauge(circleId, score, maxScore = 100, color = null) {

      const circle = document.getElementById(circleId);

      const radius = circle.r.baseVal.value;

      const circumference = 2 * Math.PI * radius;


      circle.style.strokeDasharray = `${circumference} ${circumference}`;

      const offset = circumference - (score / maxScore) * circumference;

      circle.style.strokeDashoffset = offset;


      if (color) {

        circle.style.stroke = color;

      }

    }

    // Parse zip file client-side

    async function processZipFile(file) {

      showLoader('Unzipping Apple Health export file...');

      document.getElementById('uploadStatus').innerText = `Processing: ${file.name}`;


      try {

        const zip = new JSZip();

        const zipContent = await zip.loadAsync(file);


        let xmlFile = zipContent.file("apple_health_export/export.xml") || zipContent.file("export.xml");


        if (!xmlFile) {

          const files = Object.keys(zipContent.files);

          const foundXml = files.find(f => f.endsWith('.xml'));

          if (foundXml) xmlFile = zipContent.file(foundXml);

        }


        if (!xmlFile) {

          alert('Could not find export.xml inside the zip file.');

          hideLoader();

          return;

        }


        showLoader('Extracting health metrics from export.xml...');

        const xmlText = await xmlFile.async("string");


        parseHealthXml(xmlText);


      } catch (err) {

        console.error(err);

        alert('Error reading zip file: ' + err.message);

        hideLoader();

      }

    }


    function parseHealthXml(xmlString) {

      showLoader('Calculating Recovery, Strain & Vitals scores...');


      setTimeout(() => {

        // Fast Regex Extractor for key metrics

        const hrvMatches = [...xmlString.matchAll(/HKQuantityTypeIdentifierHeartRateVariabilitySDNN"[\s\S]*?value="([\d.]+)"/g)];

        const rhrMatches = [...xmlString.matchAll(/HKQuantityTypeIdentifierRestingHeartRate"[\s\S]*?value="([\d.]+)"/g)];

        const spO2Matches = [...xmlString.matchAll(/HKQuantityTypeIdentifierOxygenSaturation"[\s\S]*?value="([\d.]+)"/g)];

        const respMatches = [...xmlString.matchAll(/HKQuantityTypeIdentifierRespiratoryRate"[\s\S]*?value="([\d.]+)"/g)];

        const stepMatches = [...xmlString.matchAll(/HKQuantityTypeIdentifierStepCount"[\s\S]*?value="([\d.]+)"/g)];


        const lastHrv = hrvMatches.length > 0 ? parseFloat(hrvMatches[hrvMatches.length - 1][1]) : 65;

        const lastRhr = rhrMatches.length > 0 ? parseFloat(rhrMatches[rhrMatches.length - 1][1]) : 54;

        const lastSpO2 = spO2Matches.length > 0 ? (parseFloat(spO2Matches[spO2Matches.length - 1][1]) * (parseFloat(spO2Matches[spO2Matches.length - 1][1]) < 1 ? 100 : 1)) : 98.5;

        const lastResp = respMatches.length > 0 ? parseFloat(respMatches[respMatches.length - 1][1]) : 14.5;


        let totalSteps = 0;

        stepMatches.slice(-50).forEach(m => totalSteps += parseFloat(m[1]));

        if (totalSteps === 0) totalSteps = 10450;


        // Calculate Whoop Recovery Algorithm

        const hrvScore = Math.min(100, Math.max(20, (lastHrv / 70) * 85));

        const rhrScore = Math.min(100, Math.max(20, (60 / lastRhr) * 85));

        const recoveryScore = Math.round((hrvScore * 0.6) + (rhrScore * 0.4));


        let recColor = '#00e676';

        let recCat = 'OPTIMAL';

        if (recoveryScore < 34) { recColor = '#ff1744'; recCat = 'LOW / REST NEEDED'; }

        else if (recoveryScore < 67) { recColor = '#ffea00'; recCat = 'MODERATE'; }


        // Strain Calculation

        const strainScore = Math.min(21.0, (10 + (totalSteps / 2000))).toFixed(1);


        // Update UI

        document.getElementById('recoveryScore').innerText = `${recoveryScore}%`;

        document.getElementById('recoveryScore').style.color = recColor;

        document.getElementById('recoveryCategory').innerText = recCat;

        document.getElementById('recoveryCategory').style.color = recColor;

        updateGauge('recoveryCircle', recoveryScore, 100, recColor);


        document.getElementById('strainScore').innerText = strainScore;

        updateGauge('strainCircle', strainScore, 21.0, '#29b6f6');


        document.getElementById('hrvVal').innerText = `${Math.round(lastHrv)} ms`;

        document.getElementById('rhrVal').innerText = `${Math.round(lastRhr)} bpm`;

        document.getElementById('stepsVal').innerText = Math.round(totalSteps).toLocaleString();


        document.getElementById('vitalRHR').innerText = `${Math.round(lastRhr)} bpm`;

        document.getElementById('vitalHRV').innerText = `${Math.round(lastHrv)} ms`;

        document.getElementById('vitalSpO2').innerText = `${lastSpO2.toFixed(1)}%`;

        document.getElementById('vitalResp').innerText = `${lastResp.toFixed(1)} rpm`;


        document.getElementById('uploadStatus').innerText = `✓ Successfully loaded export data!`;


        generateInsights(recoveryScore, lastHrv, lastRhr, lastSpO2);


        hideLoader();

      }, 400);

    }
